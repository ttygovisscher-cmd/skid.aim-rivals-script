local ACTIVE_CONFIG_NAME = "default"

local function get_config_path(name)
    return CFG_FOLDER .. "/" .. (name or "default") .. ".json"
end

local function save_config(name)
    name = name or ACTIVE_CONFIG_NAME
    local ok, enc = pcall(HttpService.JSONEncode, HttpService, Config)
    if ok and writefile then
        pcall(writefile, get_config_path(name), enc)
        pcall(writefile, CFG_FOLDER .. "/_active.txt", name)
    end
end

local function load_config(name)
    name = name or ACTIVE_CONFIG_NAME
    if not (isfile and readfile) then return false end
    local path = get_config_path(name)
    if not isfile(path) then return false end
    local ok, data = pcall(readfile, path)
    if not ok or not data then return false end
    local ok2, dec = pcall(HttpService.JSONDecode, HttpService, data)
    if not ok2 or type(dec) ~= "table" then return false end
    for k, v in pairs(dec) do
        if k ~= "keybinds" and k ~= "active_tab" then Config[k] = v end
    end
    if type(dec.keybinds) == "table" then
        for k, v in pairs(dec.keybinds) do Config.keybinds[k] = v end
    end
    ACTIVE_CONFIG_NAME = name
    return true
end

local function list_configs()
    local out = {}
    if not (isfolder and listfiles) then return {"default"} end
    local ok, files = pcall(listfiles, CFG_FOLDER)
    if not ok or not files then return {"default"} end
    for _, f in ipairs(files) do
        local name = f:match("([^/\\]+)%.json$")
        if name and name ~= "" and not name:find("^_") then
            table.insert(out, name)
        end
    end
    if #out == 0 then table.insert(out, "default") end
    return out
end

if isfile and readfile and isfile(CFG_FOLDER .. "/_active.txt") then
    local ok, n = pcall(readfile, CFG_FOLDER .. "/_active.txt")
    if ok and n and n ~= "" then ACTIVE_CONFIG_NAME = n end
end

local function make(cls, props, children)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(children or {}) do c.Parent = o end
    return o
end
local function tw(o, t, p, s, d)
    local g = TweenService:Create(o, TweenInfo.new(t, s or Enum.EasingStyle.Quart, d or Enum.EasingDirection.Out), p)
    g:Play(); return g
end
local function corner(o, r) return make("UICorner",{CornerRadius=UDim.new(0,r or 8),Parent=o}) end
local function stroke(o, c, t, tr)
    return make("UIStroke",{Color=c or BORDER,Thickness=t or 1,Transparency=tr or 0,
        ApplyStrokeMode=Enum.ApplyStrokeMode.Border,Parent=o})
end
local function grad(o, c1, c2, rot)
    return make("UIGradient",{Color=ColorSequence.new(c1,c2),Rotation=rot or 0,Parent=o})
end

local function slider(parent, label, key, min, max, suffix, cb)
    local r = make("Frame",{Size=UDim2.new(1,0,0,50),BackgroundColor3=BG_ROW,BorderSizePixel=0,Parent=parent})
    corner(r,8); stroke(r,BORDER,1,0.4)
    make("TextLabel",{
        Size=UDim2.new(1,-120,0,14), Position=UDim2.new(0,12,0,7),
        BackgroundTransparency=1, Text=label, TextColor3=TXT_MAIN,
        Font=Enum.Font.Gotham, TextSize=12,
        TextXAlignment=Enum.TextXAlignment.Left, Parent=r})
    local val = make("TextLabel",{
        Size=UDim2.new(0,58,0,17), Position=UDim2.new(1,-70,0,6),
        BackgroundColor3=Color3.fromRGB(8,8,12), BorderSizePixel=0,
        Text=tostring(Config[key])..(suffix or ""), TextColor3=accent(),
        Font=Enum.Font.GothamBold, TextSize=11, Parent=r})
    corner(val,4); stroke(val,Color3.fromRGB(45,42,62),1,0.3)
    local track = make("Frame",{
        Size=UDim2.new(1,-24,0,3), Position=UDim2.new(0,12,0,37),
        BackgroundColor3=Color3.fromRGB(36,34,52), BorderSizePixel=0, Parent=r})
    corner(track,2)
    local init = math.clamp((Config[key]-min)/(max-min),0,1)
    local fill = make("Frame",{Size=UDim2.new(init,0,1,0),BackgroundColor3=accent(),BorderSizePixel=0,Parent=track})
    corner(fill,2); grad(fill, hsv(Config.ui_hue_1,0.6,1), hsv(Config.ui_hue_2,0.6,1), 0)
    local kn = make("Frame",{
        Size=UDim2.new(0,10,0,10), Position=UDim2.new(init,-5,0.5,-5),
        BackgroundColor3=Color3.fromRGB(250,250,255), BorderSizePixel=0, Parent=track})
    corner(kn,5)
    local drag=false
    local function setX(x)
        local rel = math.clamp((x-track.AbsolutePosition.X)/track.AbsoluteSize.X,0,1)
        local v = min+(max-min)*rel
        if max-min>10 then v=math.floor(v) end
        v = math.floor(v*100)/100
        Config[key]=v; val.Text=tostring(v)..(suffix or "")
        fill.Size=UDim2.new(rel,0,1,0); kn.Position=UDim2.new(rel,-5,0.5,-5)
        if cb then cb(v) end
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            drag=true; setX(i.Position.X)
        end
    end)
    UserInput.InputChanged:Connect(function(i)
        if drag and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            setX(i.Position.X)
        end
    end)
    UserInput.InputEnded:Connect(function(i)
        if drag and (i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch) then
            drag=false; save_config()
        end
    end)
    r.MouseEnter:Connect(function() tw(r,0.15,{BackgroundColor3=BG_ROW_HOV}) end)
    r.MouseLeave:Connect(function() tw(r,0.15,{BackgroundColor3=BG_ROW}) end)
    return r
end

-- TARGET HELPERS (HEAD ONLY voor fast aim)

local function get_head(p)
    local char = p.Character
    if not char then return nil end
    local h = char:FindFirstChild("Head")
    return h and h.Position
end

local function in_fov(pos, fov)
    if not pos then return false end
    local cp = CAM.CFrame.Position
    local to_t = (pos - cp).Unit
    local dot = math.clamp(to_t:Dot(CAM.CFrame.LookVector), -1, 1)
    return math.deg(math.acos(dot)) <= fov / 2
end

local function get_target_head(fov)
    local best, best_d = nil, math.huge
    local cp = CAM.CFrame.Position
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local pos = get_head(p)
                if pos and in_fov(pos, fov) then
                    local d = (pos - cp).Magnitude
                    if d < best_d then
                        best_d = d
                        best = {player=p, pos=pos}
                    end
                end
            end
        end
    end
    return best
end

-- ============================================================
-- MOVE MOUSE (desktop) — pixel cap 120 voor fast aim
-- ============================================================
local function move_mouse(target_pos, max_px)
    if not target_pos then return end
    local cf = CAM.CFrame
    local r, u, l = cf.RightVector, cf.UpVector, cf.LookVector
    local to_t = target_pos - cf.Position
    local dist = to_t.Magnitude
    if dist < 0.1 then return end
    to_t = to_t / dist
    local xl, yl, zl = to_t:Dot(r), to_t:Dot(u), to_t:Dot(l)
    local h_ang = math.atan2(xl, zl)
    local v_ang = math.asin(math.clamp(yl, -1, 1))
    local vp = CAM.ViewportSize
    local fov_rad = math.rad(CAM.FieldOfView)
    local ppdx, ppdy = vp.X / fov_rad, vp.Y / fov_rad
    local smooth = math.max(1, Config.fastaim_smooth or 6)
    local px = (h_ang * ppdx) / smooth
    local py = -(v_ang * ppdy) / smooth
    local MAX = max_px or 80
    px = math.clamp(px, -MAX, MAX)
    py = math.clamp(py, -MAX, MAX)
    if mousemoverel then mousemoverel(px, py) end
end

-- ============================================================
-- MOVE CAMERA (mobile)
-- ============================================================
local function move_camera(target_pos, speed_factor)
    if not target_pos then return end
    local cf = CAM.CFrame
    local cam_pos = cf.Position
    local to_t = target_pos - cam_pos
    if to_t.Magnitude < 0.1 then return end
    local want = to_t.Unit
    local smooth = math.max(1, Config.fastaim_smooth or 6)
    local step = math.clamp((speed_factor or 1) / smooth, 0.05, 1)
    local new_look = cf.LookVector:Lerp(want, step)
    if new_look.Magnitude < 0.01 then return end
    CAM.CFrame = CFrame.lookAt(cam_pos, cam_pos + new_look.Unit, Vector3.new(0,1,0))
end

local function crosshair_on_head()
    local char = LP.Character
    if not char then return false end
    local ray = Ray.new(CAM.CFrame.Position, CAM.CFrame.LookVector * 600)
    local hit = Workspace:FindPartOnRayWithIgnoreList(ray, {char})
    if not hit then return false end
    local model = hit:FindFirstAncestorOfClass("Model")
    if not model then return false end
    local plr = Players:GetPlayerFromCharacter(model)
    if plr and plr ~= LP then
        local hum = model:FindFirstChildOfClass("Humanoid")
        return (hum and hum.Health > 0) and hit.Name == "Head"
    end
    return false
end

local function ensure_hover_instances()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    if hover_att and hover_att.Parent ~= hrp then
        pcall(function() hover_att:Destroy() end); hover_att = nil
    end
    if hover_lv and hover_lv.Parent ~= hrp then
        pcall(function() hover_lv:Destroy() end); hover_lv = nil
    end
    if not hover_att then
        hover_att = Instance.new("Attachment")
        hover_att.Name = "skid_hover_att"
        hover_att.Parent = hrp
    end
    if not hover_lv then
        hover_lv = Instance.new("LinearVelocity")
        hover_lv.Name = "skid_hover_lv"
        hover_lv.Attachment0 = hover_att
        hover_lv.MaxForce = math.huge
        hover_lv.VectorVelocity = Vector3.new(0, 0, 0)
        hover_lv.ForceLimitsEnabled = false
        hover_lv.Parent = hrp
    end
    return true
end

local function start_hover()
    if ensure_hover_instances() then
        hover_lv.Enabled = true
        hover_lv.VectorVelocity = Vector3.new(0, Config.hover_force / 60, 0)
        hover_active = true
    end
end

local function stop_hover()
    if hover_lv then
        hover_lv.VectorVelocity = Vector3.new(0, 0, 0)
        hover_lv.Enabled = false
    end
    hover_active = false
end

local function has_deflect_tool(p)
    local char = p.Character
    if not char then return false end
    for _, tool in ipairs(char:GetChildren()) do

end

local crosshair_layer = make("Frame",{
    Name="Crosshair", Size=UDim2.new(1,0,1,0),
    BackgroundTransparency=1, ZIndex=300, Parent=visual_gui})
local ch_parts = {}
local ch_rotate = make("Frame",{
    Size=UDim2.new(0,60,0,60), AnchorPoint=Vector2.new(0.5,0.5),
    Position=UDim2.new(0.5,0,0.5,0), BackgroundTransparency=1,
    ZIndex=301, Parent=crosshair_layer})

local function rebuild_crosshair()
    for _, p in ipairs(ch_parts) do p:Destroy() end
    ch_parts = {}
    if not Config.crosshair_enabled then return end
    local size = Config.crosshair_size
    local thick = Config.crosshair_thickness
    local gap = Config.crosshair_gap
    local c1 = hsv(Config.crosshair_hue_1, 0.7, 1)
    local c2 = hsv(Config.crosshair_hue_2, 0.7, 1)
    local outline = Config.crosshair_outline
    local style = Config.crosshair_style
    local use_grad = Config.crosshair_gradient

    local function add(w,h,x,y,c,gradient_to)
        local f = make("Frame",{
            Size=UDim2.new(0,w,0,h),
            Position=UDim2.new(0.5,-w/2+x, 0.5,-h/2+y),
            BackgroundColor3=c, BorderSizePixel=0, ZIndex=302, Parent=ch_rotate})
        if use_grad and gradient_to then
            grad(f, c, gradient_to, 0)
        end
        table.insert(ch_parts, f)
    end
    local function addol(w,h,x,y)
        if outline then add(w+2,h+2,x,y,Color3.fromRGB(0,0,0)) end
        add(w,h,x,y,c1,c2)
    end
    if style == "Cross" then
        addol(thick,size,0,-gap-size/2); addol(thick,size,0,gap+size/2)
        addol(size,thick,-gap-size/2,0); addol(size,thick,gap+size/2,0)
    elseif style == "Dot" then addol(thick+2,thick+2,0,0)
    elseif style == "X" then
        addol(thick,size,-gap*0.7,-gap*0.7); addol(thick,size,gap*0.7,-gap*0.7)
        addol(thick,size,-gap*0.7,gap*0.7); addol(thick,size,gap*0.7,gap*0.7)
    elseif style == "Circle" then
        local c = make("Frame",{
            Size=UDim2.new(0,size*2,0,size*2), AnchorPoint=Vector2.new(0.5,0.5),
            Position=UDim2.new(0.5,0,0.5,0), BackgroundTransparency=1,
            ZIndex=302, Parent=ch_rotate})
        corner(c,size)
        local cs = make("UIStroke",{Color=c1,Thickness=thick,Parent=c})
        if use_grad then
            local g = Instance.new("UIGradient")
            g.Color = ColorSequence.new(c1, c2)
            g.Parent = cs
        end
        table.insert(ch_parts, c)
    elseif style == "T-Shape" then
        addol(size,thick,0,-gap-size/2); addol(size,thick,-gap-size/2,0); addol(size,thick,gap+size/2,0)
    end
end
rebuild_crosshair()

            Config.crosshair_gradient, Config.crosshair_outline}
        local diff = false
        for i=1,#cur do if cur[i] ~= last[i] then diff = true break end end
        if diff then last = cur; rebuild_crosshair() end
    end
end)

        g.Color = ColorSequence.new(c1, c2)
        g.Rotation = (fov_phase * 30) % 360
        g.Parent = fov_stroke
        -- stroke kleur op wit zetten zodat gradient zichtbaar is
        fov_stroke.Color = Color3.fromRGB(255,255,255)
    else
        fov_stroke.Color = c1
    end

    if Config.fov_pulse then
        fov_phase = fov_phase + dt
        local p = 1 + math.sin(fov_phase*3) * 0.04
        fov_frame.Size = UDim2.new(0, r*2*p, 0, r*2*p)
    end
    if Config.fov_rotate then
        fov_rot.Rotation = (fov_phase*60) % 360
    end
end)

            end

end)

    while true do
        task.wait(0.03)
        local active = Config.vm_enabled and
            (Config.vm_color or Config.vm_neon or Config.vm_rainbow or Config.vm_material ~= "")
        if active then
            local vm = get_active_viewmodel()
            if vm then
                for _, part in ipairs(vm:GetDescendants()) do
                    if part:IsA("BasePart") then
                        if not vm_original[part] then
                            vm_original[part] = {color=part.Color, material=part.Material, transparency=part.Transparency}
                        end
                        local col
                        if Config.vm_rainbow then
                            col = hsv((tick()*Config.vm_rainbow_speed/360)%1 * 360, Config.vm_color_s, Config.vm_color_v)
                        else
                            col = hsv(Config.vm_color_h, Config.vm_color_s, Config.vm_color_v)
                        end
                        part.Color = col
                        if Config.vm_neon then
                            part.Material = Enum.Material.Neon
                            part.Transparency = Config.vm_neon_trans
                        end
                        if Config.vm_material ~= "" then
                            local ok, mat = pcall(function() return Enum.Material[Config.vm_material] end)
                            if ok and mat then part.Material = mat end
                        end
                    end
                end
            end
        else
            if next(vm_original) then
                for part, orig in pairs(vm_original) do
                    pcall(function()
                        part.Color = orig.color
                        part.Material = orig.material
                        part.Transparency = orig.transparency
                    end)
                end
                vm_original = {}
            end
        end
    end
end)

    setreadonly(mt, false)
    mt.__namecall = newcclosure(function(self, ...)
        if self == feedback_remote and getnamecallmethod() == "FireServer" then
            -- speel hit sound bij elke hit-bevestiging

        end
        return old(self, ...)

        pcall(play_hit)

    end)

else
    warn("[skid.aim] UseItemFeedback niet gehookt — hit sounds mogelijk niet werkend")
end

-- ============================================================
-- EIGEN-KILL DETECTIE
-- ============================================================
-- hou bij welke spelers je recent hebt geraakt (via crosshair on head)
local recent_hits = {}
task.spawn(function()
    while true do
        task.wait(0.1)
        local char = LP.Character
        if char then
            local ray = Ray.new(CAM.CFrame.Position, CAM.CFrame.LookVector * 600)
            local hit = Workspace:FindPartOnRayWithIgnoreList(ray, {char})
            if hit then
                local model = hit:FindFirstAncestorOfClass("Model")
                if model then
                    local plr = Players:GetPlayerFromCharacter(model)
                    if plr and plr ~= LP then
                        recent_hits[plr] = tick()
                    end
                end
            end
        end
    end
end)

local watched = {}
task.spawn(function()
    while true do
        task.wait(0.15)
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP then
                local char = p.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if hum then
                    if not watched[p] then
                        watched[p] = hum.Health
                    else
                        local prev = watched[p]
                        local now = hum.Health
                        if prev > 0 and now <= 0 then
                            local own_kill = recent_hits[p] and (tick() - recent_hits[p]) < 3
                            if own_kill then
                                recent_hits[p] = nil
                                pcall(function()
                                    on_kill({name=p.Name})
                                end)
                                -- kill sound
                                if Config.kill_sound_enabled then
                                    pcall(function()
                                        local s = Instance.new("Sound")
                                        s.SoundId = "rbxassetid://5810753587"
                                        s.Volume = Config.kill_sound_volume
                                        s.Parent = game:GetService("SoundService")
                                        s:Play()
                                        game:GetService("Debris"):AddItem(s, 5)
                                    end)
                                end
                            end
                        end
                        watched[p] = now
                    end
                end
            end
        end
    end
end)

-- AUTOLOAD 1v1
-- ============================================================
local function in_1v1()
    local mychar = LP.Character
    if not mychar then return false end
    local myhrp = mychar:FindFirstChild("HumanoidRootPart")
    if not myhrp then return false end
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp and (hrp.Position - myhrp.Position).Magnitude < 500 then
                count = count + 1
            end
        end
    end
    return count == 1
end

local was_1v1 = false
task.spawn(function()
    while true do
        task.wait(2)
        if Config.autoload_enabled and Config.autoload_on_1v1 then
            local cur = in_1v1()
            if cur and not was_1v1 then
                push_notif("1v1 detected", "TopRight", Color3.fromRGB(120,132,255))
            end
            was_1v1 = cur
        end
    end
end)
